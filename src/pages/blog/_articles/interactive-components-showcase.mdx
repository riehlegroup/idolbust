import {
  SubscriptionForm,
  TrackedFreeTextQuestion,
  TrackedPoll,
  TrackedTwoWaySelection,
} from "@/components";

This post demonstrates all new interactive components in one place. The interactive section runs entirely on the client.

In Astro MDX, function props cannot cross the server/client boundary, so these examples use tracked client components with configurable event metadata.

## Subscription form

<SubscriptionForm
  client:load
  title="Subscribe to research updates"
  description="We send one short monthly summary."
/>

## Poll

<TrackedPoll
  client:only="react"
  eventName="blog_interaction"
  optionEventKey="selectedOption"
  question="Which user research activity do you run most often?"
  options={["Interviews", "Surveys", "Usability tests"]}
/>

## Two-way selection

<TrackedTwoWaySelection
  client:only="react"
  eventName="blog_interaction"
  optionEventKey="selectedOption"
  question="Where should we focus next?"
  leftOption={{ label: "Faster recruiting", value: "recruiting" }}
  rightOption={{ label: "Better synthesis", value: "synthesis" }}
/>

## Free text

<TrackedFreeTextQuestion
  client:only="react"
  eventName="blog_interaction"
  answerLengthEventKey="responseLength"
  question="What is one improvement that would make your research process easier?"
  minLength={10}
/>
